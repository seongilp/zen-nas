# zenfs TUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** zen NAS에서 `zenfs` 한 번으로 ZFS 풀·데이터셋·스냅샷을 조회·관리하는 Go/bubbletea TUI를 만들고 이미지 빌드에 통합한다.

**Architecture:** `tui/zfs/`(CLI 실행·파싱·명령 조립, bubbletea 무관)와 `tui/ui/`(bubbletea)로 계층 분리. 변경 작업은 `Cmd` 값(명령 인자 배열)으로 조립되어 확인 모달에 전문 표시 후 실행된다. 바이너리는 git에 커밋하지 않고 build.sh가 맥에서 `GOOS=freebsd GOARCH=amd64` 크로스 컴파일해 오버레이에 주입한다.

**Tech Stack:** Go 1.22+, github.com/charmbracelet/bubbletea + lipgloss + bubbles(textinput), FreeBSD `zfs`/`zpool`/`diskinfo` CLI.

**참고 스펙:** `docs/superpowers/specs/2026-07-11-zenfs-tui-design.md`

## 전제 지식 (실행자 필독)

- 맥(호스트)에서 개발·테스트한다. 파서/ops 유닛 테스트는 전부 맥에서 돈다(픽스처 기반). 실행 환경은 FreeBSD 15 amd64 (zen 이미지).
- ZFS 머신 출력: `-H`(헤더 없음, 탭 구분) `-p`(정확한 바이트 수). 스냅샷 행에서 avail/mountpoint/compression은 `-`.
- 위험 작업 안전장치가 이 도구의 존재 이유다. `zfs/ops.go`의 명령 조립 테스트와 `ui/confirm.go`의 이름 타이핑 검증 테스트는 생략 불가.
- Go가 맥에 없으면: `brew install go`.
- 커밋은 main 브랜치가 아닌 `impl/zenfs` 브랜치에서 (Task 1에서 생성).

```go
package zfs

// Pool: zpool list 한 행.
type Pool struct {
	Name   string
	Size   uint64
	Alloc  uint64
	Free   uint64
	Health string
}

type DatasetType string

const (
	TypeFilesystem DatasetType = "filesystem"
	TypeVolume     DatasetType = "volume"
	TypeSnapshot   DatasetType = "snapshot"
)

// Dataset: zfs list 한 행 (스냅샷 포함).
type Dataset struct {
	Name        string
	Type        DatasetType
	Used        uint64
	Avail       uint64 // 스냅샷은 0
	Refer       uint64
	Mountpoint  string // 스냅샷은 ""
	Compression string // 스냅샷은 ""
}

// PoolStatus: zpool status 파싱 결과.
type PoolStatus struct {
	Name   string
	State  string
	Scan   string // "scan:" 줄 원문 (없으면 "")
	VDevs  []VDev // config 트리를 깊이와 함께 평탄화
	Errors string
}

type VDev struct {
	Depth int // config 들여쓰기 단계 (0=풀 자체)
	Name  string
	State string
}

// Disk: 풀 생성/교체 후보 디스크.
type Disk struct {
	Name      string // 예: ada1
	SizeBytes uint64
	Note      string // "" | "OS 디스크" | "<pool> 소속"
}

// Cmd: 실행 전 사용자에게 전문을 보여주기 위한 명령 값.
type Cmd struct {
	Name string
	Args []string
}

func (c Cmd) String() string {
	s := c.Name
	for _, a := range c.Args {
		s += " " + a
	}
	return s
}
```

```go
package zfs

import (
	"bytes"
	"context"
	"os/exec"
	"time"
)

// Runner는 외부 명령 실행을 추상화한다. 테스트에서 mock으로 대체.
type Runner interface {
	Run(ctx context.Context, name string, args ...string) (stdout, stderr string, err error)
}

type ExecRunner struct {
	Timeout time.Duration // 0이면 30초
}

func (r ExecRunner) Run(ctx context.Context, name string, args ...string) (string, string, error) {
	t := r.Timeout
	if t == 0 {
		t = 30 * time.Second
	}
	ctx, cancel := context.WithTimeout(ctx, t)
	defer cancel()
	cmd := exec.CommandContext(ctx, name, args...)
	var out, errb bytes.Buffer
	cmd.Stdout, cmd.Stderr = &out, &errb
	err := cmd.Run()
	return out.String(), errb.String(), err
}
```

```go
package zfs

import (
	"context"
	"fmt"
	"strconv"
	"strings"
)

// num: -p 출력의 숫자. 스냅샷의 "-"는 0.
func num(s string) (uint64, error) {
	if s == "-" {
		return 0, nil
	}
	return strconv.ParseUint(s, 10, 64)
}

func dash(s string) string {
	if s == "-" {
		return ""
	}
	return s
}

func parsePools(out string) ([]Pool, error) {
	var pools []Pool
	for _, line := range strings.Split(strings.TrimSpace(out), "\n") {
		if line == "" {
			continue
		}
		f := strings.Split(line, "\t")
		if len(f) != 5 {
			return nil, fmt.Errorf("zpool list: 예상 밖 행: %q", line)
		}
		size, err := num(f[1])
		if err != nil {
			return nil, err
		}
		alloc, err := num(f[2])
		if err != nil {
			return nil, err
		}
		free, err := num(f[3])
		if err != nil {
			return nil, err
		}
		pools = append(pools, Pool{f[0], size, alloc, free, f[4]})
	}
	return pools, nil
}

func parseDatasets(out string) ([]Dataset, error) {
	var ds []Dataset
	for _, line := range strings.Split(strings.TrimSpace(out), "\n") {
		if line == "" {
			continue
		}
		f := strings.Split(line, "\t")
		if len(f) != 7 {
			return nil, fmt.Errorf("zfs list: 예상 밖 행: %q", line)
		}
		used, err := num(f[2])
		if err != nil {
			return nil, err
		}
		avail, err := num(f[3])
		if err != nil {
			return nil, err
		}
		refer, err := num(f[4])
		if err != nil {
			return nil, err
		}
		ds = append(ds, Dataset{
			Name: f[0], Type: DatasetType(f[1]),
			Used: used, Avail: avail, Refer: refer,
			Mountpoint: dash(f[5]), Compression: dash(f[6]),
		})
	}
	return ds, nil
}

func ListPools(ctx context.Context, r Runner) ([]Pool, error) {
	out, stderr, err := r.Run(ctx, "zpool", "list", "-Hp", "-o", "name,size,alloc,free,health")
	if err != nil {
		return nil, fmt.Errorf("zpool list: %w: %s", err, stderr)
	}
	return parsePools(out)
}

func ListDatasets(ctx context.Context, r Runner) ([]Dataset, error) {
	out, stderr, err := r.Run(ctx, "zfs", "list", "-Hp",
		"-t", "filesystem,volume,snapshot",
		"-o", "name,type,used,avail,refer,mountpoint,compression")
	if err != nil {
		return nil, fmt.Errorf("zfs list: %w: %s", err, stderr)
	}
	return parseDatasets(out)
}
```

```go
package zfs

import (
	"context"
	"fmt"
	"regexp"
	"strings"
)

func parsePoolStatus(out string) (PoolStatus, error) {
	var st PoolStatus
	inConfig := false
	for _, line := range strings.Split(out, "\n") {
		trimmed := strings.TrimSpace(line)
		switch {
		case strings.HasPrefix(trimmed, "pool:"):
			st.Name = strings.TrimSpace(strings.TrimPrefix(trimmed, "pool:"))
		case strings.HasPrefix(trimmed, "state:"):
			st.State = strings.TrimSpace(strings.TrimPrefix(trimmed, "state:"))
		case strings.HasPrefix(trimmed, "scan:"):
			st.Scan = strings.TrimSpace(strings.TrimPrefix(trimmed, "scan:"))
		case strings.HasPrefix(trimmed, "errors:"):
			st.Errors = strings.TrimSpace(strings.TrimPrefix(trimmed, "errors:"))
			inConfig = false
		case strings.HasPrefix(trimmed, "config:"):
			inConfig = true
		case inConfig:
			if trimmed == "" || strings.HasPrefix(trimmed, "NAME") {
				continue
			}
			// config 행: "\t" + (깊이×2 스페이스) + name + 공백 + state + 카운터들
			body := strings.TrimPrefix(line, "\t")
			depth := (len(body) - len(strings.TrimLeft(body, " "))) / 2
			f := strings.Fields(trimmed)
			if len(f) < 2 {
				continue
			}
			st.VDevs = append(st.VDevs, VDev{Depth: depth, Name: f[0], State: f[1]})
		}
	}
	if st.Name == "" {
		return st, fmt.Errorf("zpool status: pool 이름을 찾지 못함")
	}
	return st, nil
}

func GetPoolStatus(ctx context.Context, r Runner, name string) (PoolStatus, error) {
	out, stderr, err := r.Run(ctx, "zpool", "status", name)
	if err != nil {
		return PoolStatus{}, fmt.Errorf("zpool status: %w: %s", err, stderr)
	}
	return parsePoolStatus(out)
}

func parseDiskinfo(out string) (Disk, error) {
	f := strings.Split(strings.TrimSpace(out), "\t")
	if len(f) < 3 {
		return Disk{}, fmt.Errorf("diskinfo: 예상 밖 출력: %q", out)
	}
	size, err := num(f[2])
	if err != nil {
		return Disk{}, err
	}
	return Disk{Name: f[0], SizeBytes: size}, nil
}

var rootDevRe = regexp.MustCompile(`(?m)^/dev/([a-z]+[0-9]+)p?[0-9]*\s+/\s`)

func rootDiskFromMount(out string) string {
	m := rootDevRe.FindStringSubmatch(out)
	if m == nil {
		return ""
	}
	return m[1]
}

// ListDisks: kern.disks의 각 디스크를 diskinfo로 조사하고,
// OS 디스크와 풀 소속 디스크에 Note를 단다. cd(광학)는 제외.
func ListDisks(ctx context.Context, r Runner, statuses []PoolStatus) ([]Disk, error) {
	out, stderr, err := r.Run(ctx, "sysctl", "-n", "kern.disks")
	if err != nil {
		return nil, fmt.Errorf("kern.disks: %w: %s", err, stderr)
	}
	mountOut, _, err := r.Run(ctx, "mount", "-p")
	if err != nil {
		return nil, err
	}
	rootDisk := rootDiskFromMount(mountOut)

	inPool := map[string]string{} // 디스크명 → 풀명
	for _, st := range statuses {
		for _, v := range st.VDevs {
			if v.Depth >= 1 && !strings.Contains(v.Name, "-") { // mirror-0 같은 그룹 제외
				inPool[v.Name] = st.Name
			}
		}
	}

	var disks []Disk
	for _, name := range strings.Fields(strings.TrimSpace(out)) {
		if strings.HasPrefix(name, "cd") {
			continue
		}
		dOut, _, err := r.Run(ctx, "diskinfo", name)
		if err != nil {
			continue // 조회 불가 디스크는 건너뜀
		}
		d, err := parseDiskinfo(dOut)
		if err != nil {
			continue
		}
		if name == rootDisk {
			d.Note = "OS 디스크"
		} else if pool, ok := inPool[name]; ok {
			d.Note = pool + " 소속"
		}
		disks = append(disks, d)
	}
	return disks, nil
}
```

```go
func CmdCreateDataset(name, compression string) Cmd {
	args := []string{"create"}
	if compression != "" {
		args = append(args, "-o", "compression="+compression)
	}
	return Cmd{"zfs", append(args, name)}
}

func CmdDestroyDataset(name string, recursive bool) Cmd {
	args := []string{"destroy"}
	if recursive {
		args = append(args, "-r")
	}
	return Cmd{"zfs", append(args, name)}
}

func CmdCreateSnapshot(dataset, snap string) Cmd {
	return Cmd{"zfs", []string{"snapshot", dataset + "@" + snap}}
}

// CmdRollback: -r 필수 — 대상 이후 스냅샷 삭제를 동반한다 (UI 경고 필수).
func CmdRollback(fullSnap string) Cmd {
	return Cmd{"zfs", []string{"rollback", "-r", fullSnap}}
}

func CmdDestroySnapshot(fullSnap string) Cmd {
	return Cmd{"zfs", []string{"destroy", fullSnap}}
}

// vdevType: "single" | "mirror" | "raidz1"
func CmdCreatePool(name, vdevType string, disks []string) Cmd {
	args := []string{"create", name}
	if vdevType != "single" {
		args = append(args, vdevType)
	}
	return Cmd{"zpool", append(args, disks...)}
}

func CmdDestroyPool(name string) Cmd     { return Cmd{"zpool", []string{"destroy", name}} }

	return Cmd{"zpool", []string{"replace", pool, old, new}}
```

```go
package ui

import (
	"fmt"
	"strings"

	"github.com/charmbracelet/lipgloss"
	"zenfs/zfs"
)

type nodeKind int

const (
	nodePool nodeKind = iota
	nodeDataset
	nodeSnapshot
)

type treeNode struct {
	kind     nodeKind
	name     string // 전체 이름 (tank, tank/data, tank/data@s1)
	label    string // 표시용
	depth    int
}

// buildTree: 풀·데이터셋·스냅샷을 표시 순서대로 평탄화.
// collapsed에 든 데이터셋의 스냅샷은 숨긴다.
func buildTree(pools []zfs.Pool, datasets []zfs.Dataset, collapsed map[string]bool) []treeNode {
	var nodes []treeNode
	snapsOf := map[string][]zfs.Dataset{}
	var fsList []zfs.Dataset
	for _, d := range datasets {
		if d.Type == zfs.TypeSnapshot {
			base, _, _ := strings.Cut(d.Name, "@")
			snapsOf[base] = append(snapsOf[base], d)
		} else {
			fsList = append(fsList, d)
		}
	}
	for _, p := range pools {
		nodes = append(nodes, treeNode{nodePool, p.Name,
			fmt.Sprintf("%s (%s)", p.Name, p.Health), 0})
		for _, d := range fsList {
			if d.Name != p.Name && !strings.HasPrefix(d.Name, p.Name+"/") {
				continue
			}
			depth := strings.Count(d.Name, "/") + 1
			nodes = append(nodes, treeNode{nodeDataset, d.Name,
				fmt.Sprintf("%s  %s", shortName(d.Name), human(d.Used)), depth})
			if !collapsed[d.Name] {
				for _, s := range snapsOf[d.Name] {
					_, snap, _ := strings.Cut(s.Name, "@")
					nodes = append(nodes, treeNode{nodeSnapshot, s.Name, "@" + snap, depth + 1})
				}
			}
		}
	}
	return nodes
}

func shortName(full string) string {
	if i := strings.LastIndex(full, "/"); i >= 0 {
		return full[i+1:]
	}
	return full
}

func human(b uint64) string {
	const unit = 1024
	if b < unit {
		return fmt.Sprintf("%dB", b)
	}
	div, exp := uint64(unit), 0
	for n := b / unit; n >= unit; n /= unit {
		div *= unit
		exp++
	}
	return fmt.Sprintf("%.1f%c", float64(b)/float64(div), "KMGTPE"[exp])
}

var (
	selStyle = lipgloss.NewStyle().Reverse(true)
)

func renderTree(nodes []treeNode, cursor, height int) string {
	var b strings.Builder
	start := 0
	if cursor >= height {
		start = cursor - height + 1
	}
	for i := start; i < len(nodes) && i-start < height; i++ {
		line := strings.Repeat("  ", nodes[i].depth) + nodes[i].label
		if i == cursor {
			line = selStyle.Render(line)
		}
		b.WriteString(line + "\n")
	}
	return b.String()
}
```

```go
package ui

import (
	"context"

	tea "github.com/charmbracelet/bubbletea"
	"zenfs/zfs"
)

type refreshMsg struct {
	pools    []zfs.Pool
	datasets []zfs.Dataset
	statuses map[string]zfs.PoolStatus
	err      error
}

type opDoneMsg struct{ err error }

type App struct {
	runner   zfs.Runner
	pools    []zfs.Pool
	datasets []zfs.Dataset
	statuses map[string]zfs.PoolStatus
	nodes    []treeNode
	cursor   int
	collapsed map[string]bool
	width, height int
	modal    tea.Model // nil이면 모달 없음 (confirm/form/error)
	errText  string    // 마지막 에러 (상태바 표시)
}

func NewApp(r zfs.Runner) *App {
	return &App{runner: r, collapsed: map[string]bool{}, statuses: map[string]zfs.PoolStatus{}}
}

func (a *App) Init() tea.Cmd { return a.refresh() }

func (a *App) refresh() tea.Cmd {
	r := a.runner
	return func() tea.Msg {
		ctx := context.Background()
		pools, err := zfs.ListPools(ctx, r)
		if err != nil {
			return refreshMsg{err: err}
		}
		datasets, err := zfs.ListDatasets(ctx, r)
		if err != nil {
			return refreshMsg{err: err}
		}
		statuses := map[string]zfs.PoolStatus{}
		for _, p := range pools {
			if st, err := zfs.GetPoolStatus(ctx, r, p.Name); err == nil {
				statuses[p.Name] = st
			}
		}
		return refreshMsg{pools: pools, datasets: datasets, statuses: statuses}
	}
}

func (a *App) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch m := msg.(type) {
	case tea.WindowSizeMsg:
		a.width, a.height = m.Width, m.Height
	case refreshMsg:
		if m.err != nil {
			a.errText = m.err.Error()
			return a, nil
		}
		a.pools, a.datasets, a.statuses = m.pools, m.datasets, m.statuses
		a.nodes = buildTree(a.pools, a.datasets, a.collapsed)
		if a.cursor >= len(a.nodes) {
			a.cursor = max(0, len(a.nodes)-1)
		}
	case opDoneMsg:
		a.modal = nil
		if m.err != nil {
			a.errText = m.err.Error()
		}
		return a, a.refresh()
	case tea.KeyMsg:
		if a.modal != nil {
			var cmd tea.Cmd
			a.modal, cmd = modalUpdate(a, a.modal, m)
			return a, cmd
		}
		switch m.String() {
		case "q", "ctrl+c":
			return a, tea.Quit
		case "r":
			return a, a.refresh()
		case "up", "k":
			if a.cursor > 0 {
				a.cursor--
			}
		case "down", "j":
			if a.cursor < len(a.nodes)-1 {
				a.cursor++
			}
		case "enter", " ":
			if n := a.current(); n != nil && n.kind == nodeDataset {
				a.collapsed[n.name] = !a.collapsed[n.name]
				a.nodes = buildTree(a.pools, a.datasets, a.collapsed)
			}
		default:
			return a, a.contextKey(m.String()) // Task 8·9에서 액션 연결
		}
	}
	return a, nil
}

func (a *App) current() *treeNode {
	if a.cursor < 0 || a.cursor >= len(a.nodes) {
		return nil
	}
	return &a.nodes[a.cursor]
}

// contextKey: 선택 항목 종류에 따른 액션 키. Task 8 전까지는 no-op.
func (a *App) contextKey(k string) tea.Cmd { return nil }

func max(x, y int) int { if x > y { return x }; return y }
```

```go
		pad = 1
	}
	return left + strings.Repeat(" ", pad) + right
}

func firstLine(s string) string {
	if i := strings.IndexByte(s, '\n'); i >= 0 {
		return s[:i]
	}
	return s
```

```go
import (
```

```go
// confirmModel: 위험 작업 확인 모달. 대상 이름을 정확히 타이핑해야 실행.
type confirmModel struct {
	cmd     zfs.Cmd
	target  string // 타이핑해야 하는 이름
	title   string
	warning string // 추가 경고 (롤백 등)
	input   textinput.Model
	action  func() tea.Msg // 일치 + Enter 시 실행
	done    bool
}

func newConfirm(cmd zfs.Cmd, target, title string, action func() tea.Msg) *confirmModel {
	ti := textinput.New()
	ti.Focus()
	return &confirmModel{cmd: cmd, target: target, title: title, action: action, input: ti}
}

func (c *confirmModel) Init() tea.Cmd { return nil }

func (c *confirmModel) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	if k, ok := msg.(tea.KeyMsg); ok {
		switch k.Type {
		case tea.KeyEsc:
			return c, func() tea.Msg { return opDoneMsg{} } // 취소 = 모달 닫기
		case tea.KeyEnter:
			if c.input.Value() == c.target {
				c.done = true
				return c, c.action
			}
			return c, nil // 불일치 — 무시
```

```go
func (a *App) execCmd(c zfs.Cmd) func() tea.Msg {
	r := a.runner
	return func() tea.Msg {
		return opDoneMsg{err: zfs.Execute(context.Background(), r, c)}
	}

func (a *App) contextKey(k string) tea.Cmd {
	n := a.current()
	switch k {
	case "b": // scrub 시작 (풀)
		if n != nil && n.kind == nodePool {
			return a.execCmd(zfs.CmdScrubStart(n.name)) // 비파괴 — 확인 없이 즉시
		}
	case "B": // scrub 중지
		if n != nil && n.kind == nodePool {
			return a.execCmd(zfs.CmdScrubStop(n.name))
		}
	case "D": // 파괴 (종류별)
		if n == nil {
			return nil
		}
		switch n.kind {
		case nodePool:
			c := zfs.CmdDestroyPool(n.name)
			a.modal = newConfirm(c, n.name, "풀 파괴 — 모든 데이터가 사라집니다", a.execCmd(c))
		case nodeDataset:
			c := zfs.CmdDestroyDataset(n.name, a.hasChildren(n.name))
			a.modal = newConfirm(c, n.name, "데이터셋 파괴", a.execCmd(c))
		case nodeSnapshot:
			c := zfs.CmdDestroySnapshot(n.name)
			a.modal = newConfirm(c, n.name, "스냅샷 파괴", a.execCmd(c))
		}
	case "R": // 롤백 (스냅샷)
		if n != nil && n.kind == nodeSnapshot {
			c := zfs.CmdRollback(n.name)
			m := newConfirm(c, n.name, "스냅샷 롤백", a.execCmd(c))
			m.warning = "경고: 이 스냅샷 이후에 생성된 스냅샷·데이터가 모두 삭제됩니다 (-r)."
			a.modal = m
		}
	}
	return nil
}

func (a *App) hasChildren(name string) bool {
	for _, d := range a.datasets {
		if d.Type != zfs.TypeSnapshot && strings.HasPrefix(d.Name, name+"/") {
			return true
		}
	}
	return false
}
```

```go
import (

// textForm: 이름 하나 입력받아 build(값)로 Cmd를 만들어 실행.
type textForm struct {
	title   string
	input   textinput.Model
	build   func(string) zfs.Cmd
	exec    func(zfs.Cmd) func() tea.Msg
}

func newTextForm(title, initial string, build func(string) zfs.Cmd, exec func(zfs.Cmd) func() tea.Msg) *textForm {
	ti := textinput.New()
	ti.SetValue(initial)
	ti.Focus()
	return &textForm{title: title, input: ti, build: build, exec: exec}
}

func (f *textForm) Init() tea.Cmd { return nil }

func (f *textForm) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	if k, ok := msg.(tea.KeyMsg); ok {
		switch k.Type {
		case tea.KeyEsc:
			return f, func() tea.Msg { return opDoneMsg{} }
		case tea.KeyEnter:
			v := strings.TrimSpace(f.input.Value())
			if v == "" {
				return f, nil
			}
			return f, f.exec(f.build(v))
		}
	}
	var cmd tea.Cmd
	f.input, cmd = f.input.Update(msg)
	return f, cmd
}

func (f *textForm) View() string {
	return modalStyle.Render(fmt.Sprintf("%s\n\n%s\n\n(Enter 실행 / Esc 취소)", f.title, f.input.View()))
}

// poolForm: 이름 입력 → vdev 타입 선택 → 디스크 다중 선택 → 확인 모달로 전달.
type poolForm struct {
	step     int // 0=이름, 1=타입, 2=디스크
	name     textinput.Model
	vdevIdx  int
	disks    []zfs.Disk
	selected map[int]bool
	finish   func(name, vdevType string, disks []string) tea.Cmd
	cursor   int
}

var vdevTypes = []string{"single", "mirror", "raidz1"}

func newPoolForm(disks []zfs.Disk, finish func(string, string, []string) tea.Cmd) *poolForm {
	ti := textinput.New()
	ti.Focus()
	return &poolForm{name: ti, disks: disks, selected: map[int]bool{}, finish: finish}
}

func (f *poolForm) Init() tea.Cmd { return nil }

func (f *poolForm) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	k, ok := msg.(tea.KeyMsg)
	if !ok {
		return f, nil
	}
	if k.Type == tea.KeyEsc {
		return f, func() tea.Msg { return opDoneMsg{} }
	}
	switch f.step {
	case 0:
		if k.Type == tea.KeyEnter && strings.TrimSpace(f.name.Value()) != "" {
			f.step = 1
			return f, nil
		}
		var cmd tea.Cmd
		f.name, cmd = f.name.Update(msg)
		return f, cmd
	case 1:
		switch k.String() {
		case "up", "k":
			if f.vdevIdx > 0 { f.vdevIdx-- }
		case "down", "j":
			if f.vdevIdx < len(vdevTypes)-1 { f.vdevIdx++ }
		case "enter":
			f.step = 2
		}
	case 2:
		switch k.String() {
		case "up", "k":
			if f.cursor > 0 { f.cursor-- }
		case "down", "j":
			if f.cursor < len(f.disks)-1 { f.cursor++ }
		case " ":
			if f.disks[f.cursor].Note == "" { // 사용 중 디스크는 선택 불가
				f.selected[f.cursor] = !f.selected[f.cursor]
			}
		case "enter":
			var names []string
			for i, on := range f.selected {
				if on {
					names = append(names, f.disks[i].Name)
				}
			}
			if len(names) == 0 {
				return f, nil
			}
			return f, f.finish(strings.TrimSpace(f.name.Value()), vdevTypes[f.vdevIdx], names)
		}
	}
	return f, nil
}

func (f *poolForm) View() string {
	var b strings.Builder
	b.WriteString("새 ZFS 풀 생성\n\n")
	switch f.step {
	case 0:
		fmt.Fprintf(&b, "풀 이름: %s\n(Enter 다음 / Esc 취소)", f.name.View())
	case 1:
		b.WriteString("구성 선택:\n")
		for i, v := range vdevTypes {
			cur := "  "
			if i == f.vdevIdx { cur = "> " }
			fmt.Fprintf(&b, "%s%s\n", cur, v)
		}
		b.WriteString("(Enter 다음 / Esc 취소)")
	case 2:
		b.WriteString("디스크 선택 (Space 토글, Enter 완료):\n")
		for i, d := range f.disks {
			cur := "  "
			if i == f.cursor { cur = "> " }
			mark := "[ ]"
			if f.selected[i] { mark = "[x]" }
			note := ""
			if d.Note != "" { note = "  ← " + d.Note + " (선택 불가)" }
			fmt.Fprintf(&b, "%s%s %s  %s%s\n", cur, mark, d.Name, human(d.SizeBytes), note)
		}
	}
	return modalStyle.Render(b.String())
}

func defaultSnapName() string {
	return time.Now().Format("manual-20060102-1504")
}
```
